## 代码骨架（v1.1 落地文件：Shared/ProStore.swift）

```swift
import StoreKit

// Pro 权益检查（买断/订阅任一即解锁——Transaction.currentEntitlements 一次覆盖）
actor ProStore {
    static let shared = ProStore()

    // 商品 ID（与 ASC 配置一致）
    private let foundingID = "com.zeronelabs.uricnote.pro.founding"
    private let yearlyID = "com.zeronelabs.uricnote.pro.yearly"
    private let lifetimeID = "com.zeronelabs.uricnote.pro.lifetime"

    private var updates: Task<Void, Never>? = nil

    init() {
        updates = Task { [weak self] in
            for await update in StoreKit.Transaction.updates {
                await self?.handle(update)
            }
        }
    }

    /// 当前是否 Pro（买断 or 活跃订阅——含 30 天试用期）
    func isPro() async -> Bool {
        for await result in StoreKit.Transaction.currentEntitlements {
            if case .verified(let t) = result,
               [foundingID, yearlyID, lifetimeID].contains(t.productID),
               t.revocationDate == nil {
                return true
            }
        }
        return false
    }

    /// Founding 席位是否还有（v1.1 用本地计数+审查合规说明；云计数需服务端——v1 先全量开放 Founding 价）
    /// ⚠️ 苹果规则：IAP 价格在 ASC 配置后可改，但"前 100 名"语义在纯本地 App 无法强制——
    /// 实操：Founding 价挂全量，文案写"Founding price for early users"（不写硬性 100 名上限=避免虚假宣传风险）
    func purchaseFounding() async throws -> Bool {
        guard let product = try await Product.products(for: [foundingID]).first else { return false }
        let result = try await product.purchase()
        switch result {
        case .success(let verification):
            if case .verified(let t) = verification { await t.finish(); return true }
            return false
        case .userCancelled, .pending: return false
        @unknown default: return false
        }
    }

    private func handle(_ result: VerificationResult<StoreKit.Transaction>) async {
        if case .verified(let t) = result { await t.finish() }
    }
}
```

## 边界处理（查证过的坑）

1. **买断+订阅双持有**：用户既买了断又订阅时商店不互斥——App 检测到双持有后引导去取消订阅（`AppStore.showManageSubscriptions(in:)`）
2. **退款联动**：买断被退款（REFUND 通知）→ Pro 权益消失——`Transaction.updates` 流自动处理
3. **试用转化**：年订阅 30 天试用期内 `isPro()=true`；试用期在 ASC 配置（Subscription →  introductory offer → 30 天 free）
4. **审查备注**：提审 v1.1 时 Notes 加一句 "Pro unlock uses Apple's standard IAP; founding pricing honors early adopters per App Review Guidelines 3.1.2"

## 切换触发器联动（预设目标闭环 2026-11-13 节）
